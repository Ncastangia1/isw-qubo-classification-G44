def _get_classifier(
    classifier: str,
    seed: int,
) -> ClassifierMixin:
    """
    Instantiate a supported classifier with deterministic settings.

    Parameters
    ----------
    classifier : str
        Classifier name.
    seed : int
        Random seed.

    Returns
    -------
    ClassifierMixin
        Instantiated classifier.

    Raises
    ------
    ValueError
        If classifier is not supported.
    """

        valid_names = ", ".join(sorted(CLASSIFIER_REGISTRY))

        )

    """
    Validate a training dataset.

    Parameters
    ----------
    df : pd.DataFrame
        Dataset.
    target_column : str
        Target column name.

    Raises
    ------
    ValueError
        If dataset validation fails.

def _load_model(
    model_path: str,
) -> dict[str, Any]:
    path = Path(model_path)

    if not path.exists() or not path.is_file():
        raise FileNotFoundError(
            f"Model file does not exist: {model_path}"
        )

    artifact = joblib.load(path)

    if not isinstance(artifact, dict):
        raise ValueError(
            "Loaded model artifact must be a dictionary."
        )

    required_keys = {
        "estimator",
        "classifier_name",
        "feature_names",
    }

    missing_keys = sorted(
        required_keys.difference(artifact.keys())
    )

    if missing_keys:
        raise ValueError(
            "Invalid model artifact. Missing keys: "
            f"{missing_keys}"
        )

    LOGGER.info(
        "Model artifact loaded from '%s'.",
        model_path,
    )

    return artifact


def _validate_feature_compatibility(
    model_feature_names: list[str],
    dataset_feature_names: list[str],
) -> None:
    model_set = set(model_feature_names)
    dataset_set = set(dataset_feature_names)

    missing_features = sorted(
        model_set - dataset_set
    )
    unexpected_features = sorted(
        dataset_set - model_set
    )

    errors: list[str] = []

    if len(model_feature_names) != len(dataset_feature_names):
        errors.append(
            "Feature count mismatch "
            f"(model={len(model_feature_names)}, "
            f"dataset={len(dataset_feature_names)})"
        )

    if missing_features:
        errors.append(
            f"Missing features: {missing_features}"
        )

    if unexpected_features:
        errors.append(
            f"Unexpected features: {unexpected_features}"
        )

    if not errors and model_feature_names != dataset_feature_names:
        errors.append(
            "Feature order mismatch between model and dataset."
        )

    if errors:
        message = " | ".join(errors)

        LOGGER.error(
            "Feature compatibility validation failed: %s",
            message,
        )

        raise ValueError(
            f"Feature compatibility validation failed. {message}"

    valid_training_df: pd.DataFrame,
    valid_prediction_df: pd.DataFrame,
    tmp_path: Path,
) -> None:
    train_csv = tmp_path / "train.csv"
    test_csv = tmp_path / "test.csv"

    model_path = tmp_path / "model.joblib"
    metrics_path = tmp_path / "train_metrics.json"

    predictions_path = tmp_path / "predictions.csv"
    stats_path = tmp_path / "classification_stats.json"

    valid_training_df.to_csv(train_csv, index=False)
    valid_prediction_df.to_csv(test_csv, index=False)

    train(
        classifier="random_forest",
        reducedTrain_csv=str(train_csv),
        target_column="target",
        model_path=str(model_path),
        metrics_json=str(metrics_path),
    )

    predict(
        reduced_Test_csv=str(test_csv),
        target_column="target",
        model_path=str(model_path),
        predictions_csv=str(predictions_path),
        classif_stats_json=str(stats_path),
    )

    assert predictions_path.exists()
    assert stats_path.exists()

    predictions_df = pd.read_csv(predictions_path)

    assert {
        "row_n",
        "target",
        "prediction",
        "score",
    }.issubset(predictions_df.columns)

    assert predictions_df["score"].between(0, 1).all()

    stats = json.loads(
        stats_path.read_text(encoding="utf-8")
    )

    assert "accuracy" in stats
    assert "class_0" in stats
    assert "class_1" in stats
    assert "confusion_matrix" in stats
